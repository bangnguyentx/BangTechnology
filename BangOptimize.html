1. File MainActivity.kt (Shizuku Native Bridge + Realtime RAM Service) 
package vn.bangtechnology.bangoptimize

import android.app.*
import android.content.*
import android.os.*
import android.webkit.*
import androidx.appcompat.app.AppCompatActivity
import androidx.core.app.NotificationCompat
import rikka.shizuku.Shizuku
import java.io.BufferedReader
import java.io.InputStreamReader

class MainActivity : AppCompatActivity() {
    private lateinit var webView: WebView

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        webView = WebView(this).apply {
            settings.javaScriptEnabled = true
            settings.domStorageEnabled = true
            addJavascriptInterface(BangShizukuBridge(this@MainActivity), "BangShizukuBridge")
            loadUrl("file:///android_asset/index.html")
        }
        setContentView(webView)
        startRealtimeRamNotification()
    }

    // Tạo thông báo BangOptimize • Realtime RAM cố định như ảnh mẫu
    private fun startRealtimeRamNotification() {
        val channelId = "bang_ram_monitor"
        val nm = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            nm.createNotificationChannel(NotificationChannel(channelId, "BangOptimize Realtime RAM", NotificationManager.IMPORTANCE_LOW))
        }
        val am = getSystemService(Context.ACTIVITY_SERVICE) as ActivityManager
        val memInfo = ActivityManager.MemoryInfo()
        am.getMemoryInfo(memInfo)
        val totalMb = memInfo.totalMem / (1024 * 1024)
        val usedMb = totalMb - (memInfo.availMem / (1024 * 1024))

        val notif = NotificationCompat.Builder(this, channelId)
            .setSmallIcon(android.R.drawable.ic_dialog_info)
            .setContentTitle("BangOptimize • Realtime RAM")
            .setContentText("RAM: ${usedMb} MB / ${totalMb} MB (Mốc: 4800 MB)")
            .setOngoing(true)
            .build()
        nm.notify(2008, notif)
    }

    // Phím Cứu Hộ Vật Lý: Giữ Volume Down 3 giây tự trả về màn hình gốc 1080p
    override fun onKeyLongPress(keyCode: Int, event: KeyEvent?): Boolean {
        if (keyCode == KeyEvent.KEYCODE_VOLUME_DOWN) {
            BangShizukuBridge(this).execShizuku("wm size reset && wm density reset")
            return true
        }
        return super.onKeyLongPress(keyCode, event)
    }
}

class BangShizukuBridge(private val context: Context) {
    @JavascriptInterface
    fun execShizuku(command: String): String {
        return try {
            if (Shizuku.checkSelfPermission() != android.content.pm.PackageManager.PERMISSION_GRANTED) {
                Shizuku.requestPermission(2000)
                return "(wait) Đang xin quyền Shizuku UID 2000..."
            }
            val process = Shizuku.newProcess(arrayOf("sh", "-c", command), null, null)
            val out = BufferedReader(InputStreamReader(process.inputStream)).readText().trim()
            val code = process.waitFor()
            "(exit $code) $command ✓ $out"
        } catch (e: Exception) {
            "(err) ${e.message}"
        }
    }
}
